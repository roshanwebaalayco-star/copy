Awesome — here’s a **single-page, front-end-only Replit prompt** you can paste into a brand-new “HTML/CSS/JS” Repl. It gives you:

* A **single page app** (no backend) that runs fully in the browser.
* **Template selector** for your 6 certificate PDFs (or any you upload).
* **CSV upload** with all the fields you listed.
* **QR options**: either upload **one QR image for all rows** or **auto-generate a QR per row** from a CSV column.
* **Pixel-perfect placement** using **PDF points** (1 pt = 1/72 inch) on top of your template PDF — **no distortion**.
* **Preview with grid**, draggable field boxes, precision nudging, and live config editor.
* **Per-field clear box** (wipe prefilled text area) before drawing new value.
* **Unicode font embedding** via user-uploaded TTF (so names/addresses in Indian scripts render perfectly).
* **Output** as **ZIP of individual PDFs** or **one merged PDF** — all generated client-side.

---

## What to do in Replit (quick)

1. Create a **new Repl** → “HTML, CSS, JS”.
2. **Replace the entire `index.html`** with the code below.
3. In Replit’s file tree, make a folder **`templates/`** and upload your 6 PDFs named exactly:

   * `bihar.pdf`, `bokaro.pdf`, `gram_panchayat.pdf`, `lalpania.pdf`, `nawadih.pdf`, `up.pdf`
4. (Optional but recommended) Upload fonts (e.g., **NotoSans-Regular.ttf**, **NotoSans-Bold.ttf**, or Noto Sans Devanagari) into a folder **`fonts/`**.
5. Click **Run** → open the web view → select a template → upload CSV → (optional) upload a QR image or choose “Generate from CSV” → adjust boxes if needed → **Generate**.

> CSV headers expected (you can have more):
> `category, name, sex, dob, place_of_birth, name_of_mother, aadhaar_mother, name_of_father, aadhaar_father, address_at_birth, permanent_address, registration_number, date_of_registration, date_of_issue, updated_on, qr_content`

function blobToDataURL(blob){
  return new Promise((res,rej)=>{
    const r = new FileReader();
    r.onload = ()=>res(r.result);
    r.onerror = rej;
    r.readAsDataURL(blob);
  });
}

function wrapTextByWidth(text, font, size, maxWidth){
  const words = String(text||"").split(/\s+/);
  let lines = [], current="";
  for(let w of words){
    const test = current ? current + " " + w : w;
    const width = font.widthOfTextAtSize(test, size);
    if(width <= maxWidth) current = test;
    else {
      if(current) lines.push(current);
      current = w;
    }
  }
  if(current) lines.push(current);
  return lines;
}

}

    <select>

      <option value="right" ${spec.align==="right"?"selected":""}>right</option>
    </select>

    </select>
    <div class="row">
      <button class="btn alt" data-act="focus">Focus</button>
      <button class="btn alt" data-act="delete">Delete</button>
    </div>
  `;
  row.addEventListener("input", (e)=>{
    const inputs = row.querySelectorAll("input,select");
    const [nameEl,xEl,yEl,wEl,hEl,sizeEl,fontEl,alignEl,clearEl] = inputs;
    const oldName = row.dataset.field;
    delete getTplCfg().fields[oldName];
    const ns = {
      x: +xEl.value, y:+yEl.value, w:+wEl.value, h:+hEl.value,
      size: +sizeEl.value, font: fontEl.value, align: alignEl.value, clearBox: clearEl.value==="true"
    };
    getTplCfg().fields[nameEl.value] = ns;
    row.dataset.field = nameEl.value;
    drawOverlay();
    syncEditor();
  });
  row.addEventListener("click", (e)=>{
    const t = e.target;
    if(!(t instanceof HTMLElement)) return;
    const act = t.dataset.act;
    if(act === "focus"){
      app.activeField = row.dataset.field;
      drawOverlay(true);
    } else if(act === "delete"){
      delete getTplCfg().fields[row.dataset.field];
      renderFieldsTable();
      syncEditor();
    }
  });
  return row;
}

async function loadTemplatePreview(){
  const tpl = getTplCfg();
  let srcBytes;
  if(app.templateKey==="custom"){
    if(!app.customTplBytes){ alert("Upload a custom template PDF."); return; }
    srcBytes = app.customTplBytes;
  } else {
    // fetch from path
    const res = await fetch(tpl.path);
    if(!res.ok){ alert("Template PDF not found at: "+tpl.path); return; }
    srcBytes = new Uint8Array(await res.arrayBuffer());
  }
  app.templateBytes = srcBytes;

  // PDF.js render
  const pdf = await pdfjsLib.getDocument({data:srcBytes}).promise;
  const page = await pdf.getPage(1);
  const viewport = page.getViewport({scale:1});
  app.pdfWidth = viewport.width;
  app.pdfHeight = viewport.height;

  // scale to fit canvas width
  const cw = pdfCanvas.width;
  const sc = cw/viewport.width;
  app.scale = sc;
  const ch = viewport.height * sc;
  pdfCanvas.height = Math.round(ch);

  const ctx = pdfCanvas.getContext("2d");
  const renderContext = { canvasContext: ctx, viewport: page.getViewport({scale:sc}) };
  await page.render(renderContext).promise;

  drawOverlay();
}

function drawOverlay(focus=false){
  overlay.innerHTML = "";
  overlay.style.display = "block";
  overlay.style.pointerEvents = "none";

  // grid
  const show = showGrid.checked;
  if(show){
    const grid = document.createElement("div");
    grid.style.position="absolute";
    grid.style.inset="0";
    grid.style.pointerEvents="none";
    const minor=10, major=50;
    const minorClr="rgba(255,255,255,0.15)", majorClr="rgba(255,255,255,0.35)";
    const bg = document.createElement("canvas");
    bg.width = pdfCanvas.width; bg.height = pdfCanvas.height;
    const g = bg.getContext("2d");
    g.clearRect(0,0,bg.width,bg.height);
    g.strokeStyle=minorClr; g.lineWidth=1;
    const step = minor*app.scale;
    for(let x=0;x<=bg.width;x+=step){ g.beginPath(); g.moveTo(x,0); g.lineTo(x,bg.height); g.stroke(); }
    for(let y=0;y<=bg.height;y+=step){ g.beginPath(); g.moveTo(0,y); g.lineTo(bg.width,y); g.stroke(); }
    g.strokeStyle=majorClr; g.lineWidth=1.5;
    const stepM = major*app.scale;
    for(let x=0;x<=bg.width;x+=stepM){ g.beginPath(); g.moveTo(x,0); g.lineTo(x,bg.height); g.stroke(); }
    for(let y=0;y<=bg.height;y+=stepM){ g.beginPath(); g.moveTo(0,y); g.lineTo(bg.width,y); g.stroke(); }
    const img = document.createElement("img");
    img.src = bg.toDataURL();
    img.style.position="absolute";
    img.style.inset="0";
    overlay.appendChild(img);
  }

  // boxes
  overlay.style.pointerEvents = "auto";
  const fields = getTplCfg().fields;
  for(const [name,spec] of Object.entries(fields)){
    const box = document.createElement("div");
    box.className = "box";
    box.dataset.field = name;
    const {x,y,w,h} = spec;
    const sc = app.scale;
    // convert pt to canvas pixels (origin bottom-left)
    const left = x * sc;
    const top = (app.pdfHeight - y - h) * sc;
    Object.assign(box.style,{
      left:left+"px", top:top+"px", width:(w*sc)+"px", height:(h*sc)+"px"
    });
    if(app.activeField===name) box.classList.add("active");
    // drag & resize
    box.addEventListener("mousedown", startDrag);
    const handle = document.createElement("div");
    handle.className="handle";
    box.appendChild(handle);
    handle.addEventListener("mousedown", startResize);
    overlay.appendChild(box);
  }
}

let dragInfo=null, resizeInfo=null;
function startDrag(e){
  if(e.target.classList.contains("handle")) return;
  const box = e.currentTarget;
  app.activeField = box.dataset.field;
  const rect = overlay.getBoundingClientRect();
  dragInfo = {
    field: box.dataset.field,
    startX: e.clientX, startY: e.clientY,
    boxLeft: parseFloat(box.style.left), boxTop: parseFloat(box.style.top),
    rectTop: rect.top, rectLeft: rect.left
  };
  document.addEventListener("mousemove", onDrag);
  document.addEventListener("mouseup", endDrag);
}
function onDrag(e){
  if(!dragInfo) return;
  const dx = e.clientX - dragInfo.startX;
  const dy = e.clientY - dragInfo.startY;
  const box = [...overlay.children].find(el=>el.dataset && el.dataset.field===dragInfo.field);
  const left = dragInfo.boxLeft + dx;
  const top  = dragInfo.boxTop + dy;
  Object.assign(box.style,{left:left+"px", top:top+"px"});
  // update config (canvas px -> pt)
  const sc = app.scale;
  const spec = getTplCfg().fields[dragInfo.field];
  spec.x = Math.round(parseFloat(left)/sc);
  spec.y = Math.round(app.pdfHeight - (parseFloat(top)/sc) - spec.h);
  syncEditor(false);
}
function endDrag(){
  document.removeEventListener("mousemove", onDrag);
  document.removeEventListener("mouseup", endDrag);
  dragInfo=null;
}
function startResize(e){
  const box = e.currentTarget.parentElement;
  app.activeField = box.dataset.field;
  resizeInfo = {
    field: box.dataset.field,
    startX: e.clientX, startY: e.clientY,
    startW: parseFloat(box.style.width), startH: parseFloat(box.style.height)
  };
  document.addEventListener("mousemove", onResize);
  document.addEventListener("mouseup", endResize);
  e.stopPropagation();
}
function onResize(e){
  if(!resizeInfo) return;
  const dx = e.clientX - resizeInfo.startX;
  const dy = e.clientY - resizeInfo.startY;
  const box = [...overlay.children].find(el=>el.dataset && el.dataset.field===resizeInfo.field);
  const w = Math.max(10, resizeInfo.startW + dx);
  const h = Math.max(10, resizeInfo.startH + dy);
  Object.assign(box.style,{width:w+"px", height:h+"px"});
  const sc = app.scale;
  const spec = getTplCfg().fields[resizeInfo.field];
  spec.w = Math.round(w/sc);
  spec.h = Math.round(h/sc);
  syncEditor(false);
}
function endResize(){
  document.removeEventListener("mousemove", onResize);
  document.removeEventListener("mouseup", endResize);
  resizeInfo=null;
}

document.addEventListener("keydown", (e)=>{
  if(!app.activeField) return;
  const spec = getTplCfg().fields[app.activeField];

  let step = e.shiftKey ? 10 : 1;
  let changed = false;

    e.preventDefault();
    if(e.key==="ArrowLeft"){ spec.x -= step; changed = true; }

    if(e.key==="ArrowDown"){ spec.y -= step; changed = true; }
  }
  if(changed){ drawOverlay(true); syncEditor(false); }
});

async function generateAll(asZip){
  if(!app.templateBytes){ alert("Load template preview first."); return; }
  if(!app.csvRows.length){ alert("Load CSV first."); return; }

  const PDFLib = window.PDFLib;
  const { PDFDocument, rgb } = PDFLib;

  // Prepare fonts
  const primaryBytes = app.primaryFontBytes || null; // if null, use Standard fonts
  const boldBytes = app.boldFontBytes || null;

  if(asZip){
    const zip = new JSZip();
    for(let i=0;i<app.csvRows.length;i++){
      const row = app.csvRows[i];
      const pdfBytes = await renderOne(row, primaryBytes, boldBytes);
      const safeName = (row.name||`row_${i+1}`).replace(/[^\w\s.-]/g,'').replace(/\s+/g,'_').slice(0,120);
      const fname = `${(row.category||app.templateKey||"CERT").toUpperCase()}_${safeName}.pdf`;
      zip.file(fname, pdfBytes);
    }
    const blob = await zip.generateAsync({type:"blob"});
    saveAs(blob, "certificates.zip");
  } else {
    // merged PDF
    const merged = await PDFDocument.create();
    for(let i=0;i<app.csvRows.length;i++){
      const row = app.csvRows[i];
      const pdfBytes = await renderOne(row, primaryBytes, boldBytes);
      const part = await PDFDocument.load(pdfBytes);
      const [page] = await merged.copyPages(part, [0]);
      merged.addPage(page);
    }
    const out = await merged.save({updateFieldAppearances:false});
    saveAs(new Blob([out], {type:"application/pdf"}), "certificates_merged.pdf");
  }
}

async function renderOne(row, primaryBytes, boldBytes){
  const PDFLib = window.PDFLib;
  const { PDFDocument, rgb } = PDFLib;

  // load template into fresh doc
  const templDoc = await PDFDocument.load(app.templateBytes);
  const doc = await PDFDocument.create();
  const [tplPage] = await doc.copyPages(templDoc, [0]);
  doc.addPage(tplPage);
  const page = doc.getPages()[0];
  const { width, height } = page.getSize();

  // fontkit + fonts
  doc.registerFontkit(window.fontkit);
  let fontPrimary = null, fontBold = null;
  if(primaryBytes){
    fontPrimary = await doc.embedFont(primaryBytes, { subset: true });
  }
  if(boldBytes){
    fontBold = await doc.embedFont(boldBytes, { subset: true });
  }
  if(!fontPrimary){ fontPrimary = await doc.embedStandardFont(PDFLib.StandardFonts.Helvetica); }
  if(!fontBold){ fontBold = await doc.embedStandardFont(PDFLib.StandardFonts.HelveticaBold); }

  // draw fields
  const fields = getTplCfg().fields;
  const textColor = rgb(0,0,0);
  for(const [name,spec] of Object.entries(fields)){
    const valRaw = row[name] ?? "";
    let val = String(valRaw);
    if(name==="updated_on" && !val) val = nowStamp();

    // clear box under field if requested
    if(spec.clearBox){
      page.drawRectangle({
        x: spec.x, y: spec.y, width: spec.w, height: spec.h,
        color: rgb(1,1,1), borderWidth: 0
      });
    }

    // which font
    const font = spec.font==="bold" ? fontBold : fontPrimary;
    const size = spec.size || app.config.defaultFontSize;

    // wrap for multi-line fields (addresses)
    const isMultiline = /address/.test(name);
    if(isMultiline){
      const lines = wrapTextByWidth(val, font, size, spec.w);
      const lineH = size * 1.2;
      let y = spec.y + spec.h - lineH;
      for(const line of lines){
        drawAlignedText(page, line, font, size, textColor, spec, y);
        y -= lineH;
        if(y < spec.y) break;
      }
    } else {
      // single line
      const y = spec.y + (spec.h - size)/2; // vertical center-ish
      drawAlignedText(page, val, font, size, textColor, spec, y);
    }
  }

  // draw QR
  const qrCfg = getTplCfg().qr;
  if(qrCfg){
    if($("#qrMode").value==="single"){
      if(app.qrImageBytes){
        const png = await doc.embedPng(app.qrImageBytes).catch(()=>null);
        const jpg = png ? null : await doc.embedJpg(app.qrImageBytes).catch(()=>null);
        const img = png || jpg;
        if(img){
          page.drawImage(img,{x:qrCfg.x, y:qrCfg.y, width:qrCfg.sizePt, height:qrCfg.sizePt});
        }
      }
    } else {
      // from CSV column 'qr_content'
      const text = row.qr_content || `CERT|cat=${(row.category||app.templateKey||"X")}|name=${row.name||""}|reg=${row.registration_number||""}|dob=${row.dob||""}|m4=${last4(row.aadhaar_mother||"")}|f4=${last4(row.aadhaar_father||"")}`;
      const dataUrl = await makeQrDataUrl(text);
      const pngBytes = await (await fetch(dataUrl)).arrayBuffer();
      const img = await doc.embedPng(pngBytes);
      page.drawImage(img,{x:qrCfg.x, y:qrCfg.y, width:qrCfg.sizePt, height:qrCfg.sizePt});
    }
  }

  return await doc.save({ updateFieldAppearances:false });
}

function drawAlignedText(page, text, font, size, color, spec, baselineY){
  // compute x start according to align
  let x = spec.x;
  if(spec.align==="center"){
    const w = font.widthOfTextAtSize(text, size);
    x = spec.x + (spec.w - w)/2;
  } else if(spec.align==="right"){
    const w = font.widthOfTextAtSize(text, size);
    x = spec.x + (spec.w - w);
  }
  page.drawText(String(text||""), {
    x, y: baselineY,
    size, font, color
  });
}

## Using it (step-by-step)

1. **Upload templates** in `/templates` and fonts in `/fonts` (optional).
2. **Load preview**: pick a template → click **Load Template Preview**.
3. **Fine-tune fields**: drag boxes or use arrow keys; adjust size/align; tick **clear** for boxes that cover prefilled sample text.
4. **Upload CSV**.
5. **QR**: choose **Generate from CSV** (uses `qr_content` column), or **Use one QR image** (uploads a single image for all rows).
6. **Generate ZIP** (one PDF per person) or **Merged PDF** (all in one).

Everything stays **front-end**; nothing is sent to servers.

---

## Practical notes (for perfection)

* **No distortion**: We never rasterize or rescale the template page — we import it and overlay content at pt coordinates.
* **Fonts**: Upload the exact TTF fonts you want (e.g., **Noto Sans**, **Noto Sans Devanagari**). This guarantees accurate rendering for Indian languages.
* **Addresses**: Long addresses wrap within the field width. Expand `w` and `h` if lines clip.
* **Aadhaar**: For safety and policy, prefer masking (`XXXX XXXX 1234`) or use only the last 4 digits in the visible text and in the QR payload.
* **“Updated On”**: If you leave it empty in CSV, we auto-fill with the current timestamp.
* **Per-row template**: If your CSV contains a `category` column matching template keys (`bihar`, `bokaro`, …), you can easily adapt the code to switch template per row; as written, it applies the currently selected template to all rows for simplicity.
